  <p>
    Questions about an installation, a report, or DiffCI's data handling go here. Messages are stored and
    read by the founder; nothing is emailed automatically, and an email address is optional — leave it if
    you want a reply.
  </p>
  <form id="contact-form" class="report-lookup" style="flex-direction:column;align-items:stretch;gap:.75rem">
    <input id="contact-email" type="email" placeholder="email (optional, for a reply)" maxlength="200">
    <input id="contact-repo" type="text" placeholder="owner/repository (optional)" pattern="[A-Za-z0-9._-]+/[A-Za-z0-9._-]+" maxlength="200">
    <textarea id="contact-message" required maxlength="4000" rows="8" placeholder="What happened, or what you want to know"></textarea>
    <button type="submit" class="button">Send →</button>
  </form>

<script>
  (function () {
    var form = document.getElementById('contact-form');
    var status = document.getElementById('contact-status');
    form.addEventListener('submit', function (event) {
      event.preventDefault();
      status.textContent = 'Sending…';
      var body = {
        email: document.getElementById('contact-email').value.trim() || undefined,
        repository: document.getElementById('contact-repo').value.trim() || undefined,
        message: document.getElementById('contact-message').value.trim()
      };
      fetch('https://diffci-research-sandbox.damp-waterfall-0cd8.workers.dev/v1/contact', {
        method: 'POST', headers: { 'content-type': 'application/json' }, body: JSON.stringify(body)
      }).then(function (r) { return r.json(); }).then(function (j) {
        status.textContent = j && j.ok ? 'Received. Thank you — the founder reads every message.' : ('Not sent: ' + ((j && j.error) || 'unknown error'));
        if (j && j.ok) form.reset();
      }).catch(function () { status.textContent = 'Not sent: network error. Please try again.'; });
    });
  })();
</script>
